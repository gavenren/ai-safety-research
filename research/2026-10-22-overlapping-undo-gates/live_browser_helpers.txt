async function saveUI(label,tab,role='audit'){const state=await tab.getAXState({emit:false,disableDiffing:true});const record={time:new Date().toISOString(),role,url:await tab.url(),state};await fsPub.writeFile(liveBase+'/private/ui-'+label+'.json',JSON.stringify(record,null,2));return state;}

async function renamePending(alias,from,to,label){const oldName='FIVE-LIVE-20261022-Q7K '+alias+' '+from,newName='FIVE-LIVE-20261022-Q7K '+alias+' '+to;const before=await saveUI(label+'-before',liveDrive);if(!before.includes(oldName))throw Error('Old name missing');await fsPub.writeFile(liveBase+'/private/dispatch-'+label+'.json',JSON.stringify({time:new Date().toISOString(),alias,oldName,newName,action:'rename'}));await liveDrive.playwright.getByRole('row',{name:new RegExp('^'+oldName+' Folder')}).click({button:'right'});await saveUI(label+'-menu',liveDrive);await liveDrive.playwright.getByRole('menuitem',{name:/^Rename/}).click();await saveUI(label+'-edit',liveDrive);await liveDrive.playwright.getByRole('textbox',{name:'Rename',exact:true}).fill(newName);await liveDrive.playwright.getByRole('button',{name:'OK',exact:true}).click();await saveUI(label+'-sent',liveDrive);await liveDrive.reload();await saveUI(label+'-reload-initial',liveDrive);return{label,status:'pending_readback'};}

async function readGateRows(label,role){const state=await saveUI(label,liveDrive,role);const out={};for(const op of ['A','B']){const loc=liveDrive.playwright.getByRole('row',{name:new RegExp('^FIVE-LIVE-20261022-Q7K GATE-'+op+' (DONE|PENDING) Folder')});const row=await loc.evaluate(el=>({id:el.getAttribute('data-id'),text:el.innerText}));if(row.id!==gateIds[op])throw Error('Gate identity mismatch');const match=row.text.match(/GATE-[AB] (DONE|PENDING)/);if(!match)throw Error('Gate name parser failed');out[op]={id:row.id,work:match[1],text:row.text};}await fsPub.writeFile(liveBase+'/private/gate-read-'+label+'.json',JSON.stringify({time:new Date().toISOString(),role,rows:out},null,2));return out;}

async function finishGateBoundary(c,step){const label=c.id+'-'+step.n;const rows=await readGateRows(label+'-audit','audit');let guarded=null;if(!step.closed)guarded=await readGateRows(label+'-guarded','guarded_read');return {time:new Date().toISOString(),case:c.id,...step,work:{A:rows.A.work,B:rows.B.work},guarded_read_completed:guarded!==null};}

async function gateDo(i,n){const c=gateCommands[i],s=c.steps[n-1];await fsPub.writeFile(liveBase+'/private/decision-'+c.id+'-'+n+'.json',JSON.stringify({time:new Date().toISOString(),...s}));if(!s.duplicate)return await renamePending('GATE-'+s.op,s.action==='C'?'DONE':'PENDING',s.action==='C'?'PENDING':'DONE',c.id+'-'+n);return{case:c.id,n,duplicateLocalOnly:true};}

async function gateObserve(i,n){const c=gateCommands[i];const rec=await finishGateBoundary(c,c.steps[n-1]);if(!gateProgress[c.id])gateProgress[c.id]=[];gateProgress[c.id].push(rec);await fsPub.writeFile(liveBase+'/private/gate-case-'+c.id+'.json',JSON.stringify(gateProgress[c.id],null,2));return{case:c.id,n,work:rec.work,closed:rec.closed,read:rec.guarded_read_completed};}

async function openLiveCal(alias,label){const m=liveManifest['calendar:'+alias];if(!m)throw Error('Unlisted fixture');await liveCal.goto(m.url);let state=await saveUI(label+'-open',liveCal);if(!state.includes('Edit event'))state=await saveUI(label+'-ready',liveCal);if(!state.includes('FIVE-LIVE-20261022-Q7K '+alias))throw Error('Event identity title absent');await liveCal.playwright.getByRole('button',{name:'Edit event',exact:true}).click();state=await saveUI(label+'-editor',liveCal);if(!state.includes('Description: Title'))state=await saveUI(label+'-editor-ready',liveCal);return state;}

async function changeLiveCal(alias,field,value,label){const state=await openLiveCal(alias,label);if(!state.includes('Description: All day, Value: 1'))throw Error('All-day contract not visible');if(!state.includes('Value: Free')||!state.includes('Value: Private'))throw Error('Privacy/free contract not visible');const fields={title:['textbox','Title'],location:['combobox','Add location'],description:['textbox','Description']};if(!fields[field])throw Error('Unlisted field');const [role,name]=fields[field];await fsPub.writeFile(liveBase+'/private/dispatch-'+label+'.json',JSON.stringify({time:new Date().toISOString(),alias,field,value,url:await liveCal.url()}));await liveCal.playwright.getByRole(role,{name,exact:true}).fill(value);await liveCal.playwright.getByRole('button',{name:'Save',exact:true}).click();await saveUI(label+'-sent',liveCal);return {label,alias,field,value,saveDispatched:true};}
