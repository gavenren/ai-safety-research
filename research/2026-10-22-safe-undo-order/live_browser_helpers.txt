async function orderReadCurrent(alias,n){const state=await saveUI(alias+'-'+n+'-editor',liveCal);const title=await liveCal.playwright.getByRole('textbox',{name:'Title',exact:true}).evaluate(el=>el.value);const location=await liveCal.playwright.getByRole('combobox',{name:'Add location',exact:true}).evaluate(el=>el.value);const description=await liveCal.playwright.getByRole('textbox',{name:'Description',exact:true}).evaluate(el=>el.innerText);const rec={time:new Date().toISOString(),editor_url:await liveCal.url(),title,location,description,all_day:state.includes('Description: All day, Value: 1'),free:state.includes('Value: Free'),private:state.includes('Value: Private')};await fsPub.writeFile(liveBase+'/private/order-read-'+alias+'-'+n+'.json',JSON.stringify(rec,null,2));return rec;}

async function orderWrite(i,n){const c=orderCommands[i],before=c.path[n-1],after=c.path[n];const b=[0,1,2].find(k=>before[k]!==after[k]);const fields=[['textbox','Title','FIVE-LIVE-20261022-Q7K '+c.alias+' TEST-A0'],['combobox','Add location','TEST-B0'],['textbox','Description','TEST-C0']];const [role,name,value]=fields[b];await fsPub.writeFile(liveBase+'/private/order-decision-'+c.alias+'-'+n+'.json',JSON.stringify({time:new Date().toISOString(),alias:c.alias,rule:c.rule,before,after,field:'abc'[b],value}));await liveCal.playwright.getByRole(role,{name,exact:true}).fill(value);await liveCal.playwright.getByRole('button',{name:'Save',exact:true}).click();await saveUI(c.alias+'-'+n+'-sent',liveCal);return{alias:c.alias,n,status:'pending_readback'};}

async function reopenOrder(alias,n){const eid=new URL(liveManifest['calendar:'+alias].url).searchParams.get('eid');await liveCal.goto('https://calendar.google.com/calendar/u/0/r/eventedit/'+eid);await liveCal.getAXState({emit:false});await liveCal.playwright.getByRole('textbox',{name:'Title',exact:true}).waitFor({state:'visible',timeoutMs:20000});return await orderReadCurrent(alias,n);}

async function checkedOrderRead(i,n){const c=orderCommands[i],r=await reopenOrder(c.alias,n),s=c.path[n];if(r.title!=='FIVE-LIVE-20261022-Q7K '+c.alias+' TEST-A'+s[0]||r.location!=='TEST-B'+s[1]||r.description!=='TEST-C'+s[2]||!r.all_day||!r.free||!r.private)throw Error('Order saved-state check failed');return {alias:c.alias,n,state:s,fieldsVerified:true};}
