async function saveUI(label,tab,role='audit'){const state=await tab.getAXState({emit:false,disableDiffing:true});const record={time:new Date().toISOString(),role,url:await tab.url(),state};await fsPub.writeFile(liveBase+'/private/ui-'+label+'.json',JSON.stringify(record,null,2));return state;}

async function renamePending(alias,from,to,label){const oldName='FIVE-LIVE-20261022-Q7K '+alias+' '+from,newName='FIVE-LIVE-20261022-Q7K '+alias+' '+to;const before=await saveUI(label+'-before',liveDrive);if(!before.includes(oldName))throw Error('Old name missing');await fsPub.writeFile(liveBase+'/private/dispatch-'+label+'.json',JSON.stringify({time:new Date().toISOString(),alias,oldName,newName,action:'rename'}));await liveDrive.playwright.getByRole('row',{name:new RegExp('^'+oldName+' Folder')}).click({button:'right'});await saveUI(label+'-menu',liveDrive);await liveDrive.playwright.getByRole('menuitem',{name:/^Rename/}).click();await saveUI(label+'-edit',liveDrive);await liveDrive.playwright.getByRole('textbox',{name:'Rename',exact:true}).fill(newName);await liveDrive.playwright.getByRole('button',{name:'OK',exact:true}).click();await saveUI(label+'-sent',liveDrive);await liveDrive.reload();await saveUI(label+'-reload-initial',liveDrive);return{label,status:'pending_readback'};}

async function calendarTokenSave(alias,from,to,label,receipt=false){const m=liveManifest['calendar:'+alias],eid=new URL(m.url).searchParams.get('eid');await liveCal.goto('https://calendar.google.com/calendar/u/0/r/eventedit/'+eid);await liveCal.getAXState({emit:false});await liveCal.playwright.getByRole('textbox',{name:'Title',exact:true}).waitFor({state:'visible',timeoutMs:20000});const state=await saveUI(label+'-before',liveCal);const title=await liveCal.playwright.getByRole('textbox',{name:'Title',exact:true}).evaluate(el=>el.value),description=await liveCal.playwright.getByRole('textbox',{name:'Description',exact:true}).evaluate(el=>el.innerText);if(title!==m.args.title||description!==from||!state.includes('Description: All day, Value: 1')||!state.includes('Value: Free')||!state.includes('Value: Private'))throw Error('Calendar before-field contract failed');await liveCal.playwright.getByRole('textbox',{name:'Description',exact:true}).fill(to);const filled=await saveUI(label+'-filled',liveCal);const actual=await liveCal.playwright.getByRole('textbox',{name:'Description',exact:true}).evaluate(el=>el.innerText);const url=await liveCal.url(),observed_id=Buffer.from(url.split('/eventedit/')[1],'base64').toString('utf8').split(' ')[0];if(actual!==to||observed_id!==m.id)throw Error('Calendar identity or intended field mismatch');if(receipt)await fsPub.writeFile(liveBase+'/private/intent-ui-calendar-'+alias+'.json',JSON.stringify({time:new Date().toISOString(),service:'calendar',alias,url,observed_id,values:{description:actual,all_day:filled.includes('Description: All day, Value: 1'),availability:filled.includes('Value: Free')?'Free':'unknown',visibility:filled.includes('Value: Private')?'Private':'unknown'},source_file:'ui-'+label+'-filled.json'},null,2));await fsPub.writeFile(liveBase+'/private/dispatch-'+label+'.json',JSON.stringify({time:new Date().toISOString(),label,alias,field:'description',from,to,url}));await liveCal.playwright.getByRole('button',{name:'Save',exact:true}).click();await saveUI(label+'-sent',liveCal);return{alias,to,status:'pending_connector_read'};}
